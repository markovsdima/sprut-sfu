🚀 SFU Сервер API Документация

WebSocket: ws://localhost:8080/ws?peerId=YOUR_PEER_ID

JSON-RPC 2.0 протокол для WebRTC групповых звонков

📋 Общая информация

Протокол

Все сообщения используют JSON-RPC 2.0 формат:

{
  "jsonrpc": "2.0",
  "method": "METHOD_NAME",
  "params": { ... },
  "id": 123
}

Статус-коды ошибок

Код Описание Тип
-32700 Ошибка парсинга JSON Стандартная
-32600 Неверный запрос Стандартная
-32601 Метод не найден Стандартная
-32602 Неверные параметры Стандартная
-32000 Комната не найдена SFU
-32001 Комната заполнена SFU
-32002 Участник не найден SFU
-32003 Уже в комнате SFU
-32004 Не в комнате SFU
-32005 Ошибка SDP SFU
-32006 Ошибка ICE SFU
-32007 Ошибка медиапотока SFU

🔗 Методы (Клиент → Сервер)

room.join обязательный

Присоединиться к комнате для группового звонка

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "room.join",
  "params": {
    "roomId": "test-room",
    "peerId": "user-123",
    "displayName": "Иван Петров",
    "auth": ""
  },
  "id": 1
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "success": true,
    "peers": ["user-456", "user-789"]
  },
  "id": 1
}

room.leave опциональный

Покинуть комнату

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "room.leave",
  "params": {
    "roomId": "test-room"
  },
  "id": 2
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "success": true
  },
  "id": 2
}

publisher.offer обязательный

Отправить SDP предложение для публикации медиа (аудио/видео)

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "publisher.offer",
  "params": {
    "sdp": "v=0\r\no=- 123456789 1 IN IP4 0.0.0.0\r\n...",
    "type": "offer"
  },
  "id": 3
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "sdp": "v=0\r\no=- 123456789 2 IN IP4 192.168.1.100\r\n...",
    "type": "answer"
  },
  "id": 3
}

subscriber.answer ответ

Отправить SDP ответ на предложение сервера для получения медиа

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "subscriber.answer",
  "params": {
    "peerId": "user-456",
    "sdp": "v=0\r\no=- 123456789 2 IN IP4 192.168.1.100\r\n...",
    "type": "answer"
  },
  "id": 4
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "success": true
  },
  "id": 4
}

ice.candidate ответ

Отправить ICE кандидата для установления соединения

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "ice.candidate",
  "params": {
    "target": "publisher",
    "peerId": "user-456",
    "candidate": "candidate:1 1 UDP 2130706431 192.168.1.100 54400 typ host",
    "sdpMid": "0",
    "sdpMLineIndex": 0
  },
  "id": 5
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "success": true
  },
  "id": 5
}

camera.state опциональный

Уведомить сервер о состоянии камеры (включена/отключена) для оптимизации трафика

Запрос: 
{
  "jsonrpc": "2.0",
  "method": "camera.state",
  "params": {
    "enabled": true
  },
  "id": 6
}
Ответ: 
{
  "jsonrpc": "2.0",
  "result": {
    "success": true
  },
  "id": 6
}
💡 Оптимизация: При отключении камеры сервер перестает ретранслировать видео пакеты другим участникам, экономя трафик и ресурсы CPU.

📢 Уведомления (Сервер → Клиент)

subscriber.offer

Сервер отправляет предложение для получения медиа от других участников

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "subscriber.offer",
  "params": {
    "sdp": "v=0\r\no=- 123456789 2 IN IP4 192.168.1.100\r\n..."
  }
}

ice.candidate

Сервер отправляет ICE кандидата для соединения

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "ice.candidate",
  "params": {
    "target": "publisher",
    "candidate": "candidate:2 1 UDP 1694498815 203.0.113.100 54401 typ srflx",
    "sdpMid": "0",
    "sdpMLineIndex": 0
  }
}

peer.joined

Новый участник присоединился к комнате

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "peer.joined",
  "params": {
    "peerId": "user-456",
    "displayName": "Мария Иванова"
  }
}

peer.left

Участник покинул комнату

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "peer.left",
  "params": {
    "peerId": "user-456"
  }
}

camera.state.changed

Уведомление об изменении состояния камеры участника

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "camera.state.changed",
  "params": {
    "peerId": "user-456",
    "enabled": false
  }
}

error

Ошибка (peer_disconnected, room_full и т.д.)

Уведомление: 
{
  "jsonrpc": "2.0",
  "method": "error",
  "params": {
    "code": -32004,
    "message": "not in room"
  }
}

💻 Пример использования (JavaScript)

Полный пример WebRTC клиента: 
// Подключение к SFU серверу
const ws = new WebSocket('ws://localhost:8080/ws?peerId=user123');

// Обработка входящих сообщений
ws.onmessage = async (event) => {
  const msg = JSON.parse(event.data);
  
  if (msg.method === 'subscriber.offer') {
    // Обработать предложение медиа соединения
    await handleSubscriberOffer(msg.params.sdp);
  } else if (msg.method === 'peerJoined') {
    // Показать нового участника
    showPeerJoined(msg.params);
  }
};

// Присоединение к комнате
function joinRoom() {
  ws.send(JSON.stringify({
    jsonrpc: "2.0",
    method: "room.join",
    params: {
      roomId: "test-room",
      peerId: "user123",
      displayName: "Моё Имя"
    },
    id: 1
  }));
}

// Отправка медиа
async function sendMediaOffer() {
  const pc = new RTCPeerConnection({ iceServers: [...] });
  const offer = await pc.createOffer();
  
  ws.send(JSON.stringify({
    jsonrpc: "2.0",
    method: "publisher.offer",
    params: {
      sdp: offer.sdp,
      type: offer.type
    },
    id: 2
  }));
}

// Управление камерой
function toggleCamera() {
  const videoTrack = localStream.getVideoTracks()[0];
  if (videoTrack) {
    videoTrack.enabled = !videoTrack.enabled;
    
    // Уведомляем сервер для оптимизации трафика
    ws.send(JSON.stringify({
      jsonrpc: "2.0",
      method: "camera.state",
      params: {
        enabled: videoTrack.enabled
      },
      id: 3
    }));
  }
}

⚙️ Конфигурация сервера

config.yaml: 
server:
  host: "0.0.0.0"
  port: 8080
  log_level: "debug"

webrtc:
  ice_servers:
    - urls: ["stun:stun.l.google.com:19302"]
    - urls: ["stun:stun1.l.google.com:19302"]
    - urls: ["stun:stun2.l.google.com:19302"]
    - urls: ["stun:stun3.l.google.com:19302"]

rooms:
  max_participants: 100
  idle_timeout: "30m"
  cleanup_interval: "5m"

environment: "development"
⚠️ Важно: Для продакшена настройте TURN серверы в конфиге для обхода NAT ограничений.
